# XAMPP Virtual Hosts Configuration for NeoCafe Domain Routing
# Add this to: C:\xampp\apache\conf\extra\httpd-vhosts.conf

# User Domain - Points to main NeoCafe directory
<VirtualHost *:8080>
    DocumentRoot "C:/xampp/htdocs/NeoCafe"
    ServerName neocafe.cafe
    <Directory "C:/xampp/htdocs/NeoCafe">
        AllowOverride All
        Require all granted
        Options Indexes FollowSymLinks MultiViews
    </Directory>
    
    # Log files
    ErrorLog "logs/neocafe.cafe-error.log"
    CustomLog "logs/neocafe.cafe-access.log" combined
</VirtualHost>

# Admin Domain - Points to main NeoCafe directory (same as user domain)
<VirtualHost *:8080>
    DocumentRoot "C:/xampp/htdocs/NeoCafe"
    ServerName admin.neocafe.cafe
    <Directory "C:/xampp/htdocs/NeoCafe">
        AllowOverride All
        Require all granted
        Options Indexes FollowSymLinks MultiViews
    </Directory>
    
    # Log files
    ErrorLog "logs/admin.neocafe.cafe-error.log"
    CustomLog "logs/admin.neocafe.cafe-access.log" combined
</VirtualHost>
